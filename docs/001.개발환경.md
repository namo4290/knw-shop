# 1. 빈 backend 폴더로 이동 (이미 있다면 삭제 후 다시 생성 권장)

cd ~/knw/knw-shop
rm -rf backend
mkdir backend

# 2. 인스톨러로 직접 생성 (현재 폴더에 생성)

laravel new backend --no-interaction

# 1. 컨테이너 빌드 및 실행

cd ~/knw/knw-shop/backend

# 현재 폴더(.)에 라라벨 13 설치

docker run --rm -v $(pwd):/app -w /app composer:latest composer create-project laravel/laravel . "^13.0"

# 컨테이너 실행

docker-compose up -d --build

# 소유권 변경 (중요: WSL2와 Docker 간 권한 충돌 방지)

sudo chown -R $USER:www-data .
chmod -R 775 storage bootstrap/cache

# 필라멘트 5 및 의존성 설치

docker-compose exec app composer require filament/filament:^5.0 -W

# 관리자 패널 설치

docker-compose exec app php artisan filament:install --panels

# 모든 캐시 날리기

docker-compose exec app php artisan optimize:clear

# 필라멘트 에셋 갱신 및 컴포넌트 캐시 정리

docker-compose exec app php artisan filament:upgrade
docker-compose exec app php artisan filament:assets

# 4. 프론트엔드 빌드 캐시 삭제 (Vite) public/build 폴더를 날리고 다시 빌드

rm -rf public/build
docker-compose exec app npm run build

# ㅡ컴포저(Composer) 및 패키지 최적화

docker-compose exec app composer dump-autoload
